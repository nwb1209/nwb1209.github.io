核心代码：前向传播 & 反向传播 SGD

// 前向传播
function forward(x, W1, b1, W2, b2){
    const h = sigmoid(matMul(x,W1).add(b1));
    const y = sigmoid(matMul(h,W2).add(b2));
    return {h,y};
}
// 反向传播、梯度更新（SGD）
function back(x, y_true, cache, W1,b1,W2,b2,lr){
    const {h,y}=cache;
    const dy = y.sub(y_true).mul(2);
    const dW2 = matMul(h.T(), dy.mul(sigmoidGrad(y)));
    const db2 = dy.mul(sigmoidGrad(y));
    const dh = matMul(dy.mul(sigmoidGrad(y)), W2.T());
    const dW1 = matMul(x.T(), dh.mul(sigmoidGrad(h)));
    const db1 = dh.mul(sigmoidGrad(h));
    // 参数更新
    W1 = W1.sub(dW1.mul(lr));
    b1 = b1.sub(db1.mul(lr));
    W2 = W2.sub(dW2.mul(lr));
    b2 = b2.sub(db2.mul(lr));
    return {W1,b1,W2,b2};
}

结果分析

1. 随着训练迭代，loss持续下降，模型逐步拟合XOR非线性；
2. 权重热力图变化：参数从随机初始化，逐步收敛到合适分布；
3. 学习率太大容易震荡不收敛，太小收敛速度很慢。
