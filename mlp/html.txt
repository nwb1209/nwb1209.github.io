<script>
// 训练数据集：[糖量,奶量] → 甜度 0~1
const trainData = [
  {x:[0.1,0.8], y:0.10},
  {x:[0.9,0.2], y:0.95},
  {x:[0.8,0.7], y:0.45},
  {x:[0.2,0.1], y:0.30},
  {x:[0.9,0.9], y:0.35},
  {x:[0.5,0.2], y:0.70},
  {x:[0.3,0.6], y:0.25},
  {x:[0.7,0.3], y:0.82},
]
let W1,b1,W2,b2
let lossList = []
const hiddenNum = 4
const lr = 0.4
const totalEpoch = 4000

// 激活函数
function sigmoid(x){ return 1/(1+Math.exp(-x)) }
function dsigmoid(y){ return y*(1-y) }

// 初始化权重
function init(){
  W1 = Array(2).fill().map(()=>Array(hiddenNum).fill().map(()=>Math.random()*2-1))
  b1 = Array(hiddenNum).fill().map(()=>Math.random()*2-1)
  W2 = Array(hiddenNum).fill().map(()=>[Math.random()*2-1])
  b2 = [Math.random()*2-1]
  lossList = []
  document.getElementById('paramText').innerText = "初始化完成，等待训练"
}

// 前向传播
function forward(x){
  let z1 = [
    x[0]*W1[0][0] + x[1]*W1[1][0] + b1[0],
    x[0]*W1[0][1] + x[1]*W1[1][1] + b1[1],
    x[0]*W1[0][2] + x[1]*W1[1][2] + b1[2],
    x[0]*W1[0][3] + x[1]*W1[1][3] + b1[3]
  ]
  let a1 = z1.map(sigmoid)
  let z2 = a1[0]*W2[0][0] + a1[1]*W2[1][0] + a1[2]*W2[2][0] + a1[3]*W2[3][0] + b2[0]
  let sweet = sigmoid(z2)
  return {a1, sweet, z2}
}

// 训练
async function startTrain(){
  init()
  for(let e=0;e<totalEpoch;e++){
    let totalLoss = 0
    for(let item of trainData){
      let x = item.x
      let yTrue = item.y
      let {a1, sweet, z2} = forward(x)
      let err = sweet - yTrue
      totalLoss += err*err
      // 反向传播
      let s = sigmoid(z2)
      let d2 = err * dsigmoid(s)
      for(let j=0;j<hiddenNum;j++){
        W2[j][0] -= lr * d2 * a1[j]
      }
      b2[0] -= lr * d2
      let d1 = []
      for(let j=0;j<hiddenNum;j++){
        d1[j] = d2 * W2[j][0] * dsigmoid(a1[j])
      }
      W1[0][0] -= lr*d1[0]*x[0]
      W1[1][0] -= lr*d1[0]*x[1]
      W1[0][1] -= lr*d1[1]*x[0]
      W1[1][1] -= lr*d1[1]*x[1]
      W1[0][2] -= lr*d1[2]*x[0]
      W1[1][2] -= lr*d1[2]*x[1]
      W1[0][3] -= lr*d1[3]*x[0]
      W1[1][3] -= lr*d1[3]*x[1]
      b1[0] -= lr*d1[0]
      b1[1] -= lr*d1[1]
      b1[2] -= lr*d1[2]
      b1[3] -= lr*d1[3]
    }
    lossList.push(totalLoss / trainData.length)
    if(e%80===0){
      drawAll()
      showParam(e)
      await sleep(5)
    }
  }
  drawAll()
